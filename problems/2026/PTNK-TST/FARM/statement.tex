\makeproblem{FARM}{Bài 1}{1.0s}{1024 MB}

Cho dãy gồm $n$ số nguyên $a_1, a_2, \dots, a_n$. Hãy chia dãy số này thành nhiều đoạn con liên tiếp nhất sao cho:

\begin{itemize}
    \item Mỗi phần tử phải nằm trong đúng một đoạn.
    \item Tổng các phần tử trong mỗi đoạn con đều là số dương.
    \item Tổng các phần tử của đoạn con đứng trước không vượt quá tổng các phần tử của đoạn con đứng sau.
\end{itemize}

\noindent Cụ thể, tìm số nguyên $k$ lớn nhất sao cho tồn tại các vị trí cắt $0 = x_0 < x_1 < x_2 < \dots < x_{k-1} < x_k = n$ thỏa mãn điều kiện sau:
\begin{enumerate}
    \item $\sum_{j=x_{i-1}+1}^{x_i} a_j > 0 \quad \forall 1 \le i \le k$.
    \item $\sum_{j=x_{i-1}+1}^{x_i} a_j \le \sum_{j=x_{i}+1}^{x_{i+1}} a_j \quad \forall 1 \le i < k$.
\end{enumerate}

\makesection{Input}
\begin{itemize}
    \item Dòng đầu tiên chứa số nguyên $n \, (1 \le n \le 10^{5})$.
    \item Dòng tiếp theo chứa $n$ số nguyên $a_1, a_2, \dots, a_n \, \left(\sum{a_i} > 0, \sum{|a_i|} \le 10^{5}\right)$.
\end{itemize}

\makesection{Output}
\begin{itemize}
    \item In ra một số nguyên duy nhất là số đoạn nhiều nhất chia được.
\end{itemize}

\begin{makesubtasks}
    \subtask{20} $n \le 20$.
    \subtask{20} $n \le 700$.
    \subtask{20} $n \le 2500$.
    \subtask{20} $\sum{|a_i|} \le 10^{4}$.
    \subtask{20} Không có ràng buộc gì thêm.
\end{makesubtasks}
