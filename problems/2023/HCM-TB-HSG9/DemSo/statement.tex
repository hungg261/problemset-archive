\makeproblem{DemSo}{Đếm số (5đ)}{1.0s}{256 MB}

Nhập một dãy $A$ có $N$ số tự nhiên $(N < 40)$ và 1 số $k$. Hãy xuất ra các phần tử có số lần xuất hiện trong dãy $A$ từ $K$ lần trở lên (Mỗi số chỉ xuất 1 lần).

\makesection{Input}
\begin{itemize}
    \item Gồm một dòng duy nhất chứa số $N, K$ cách nhau một khoảng trắng.
    \item Dòng tiếp theo chứa $N$ số nguyên, mối số cách nhau một khoảng trắng lần lượt là các phần tử trong dãy $A$.
\end{itemize}

\makesection{Output}
\begin{itemize}
    \item In ra các số thỏa điều kiện trên, trường hợp không có số nào thỏa thì xuất số $-1$.
\end{itemize}

\makesection{Ví dụ}

\subsubsection*{Ví dụ 1}
\begin{makesample}
\begin{lstlisting}
8 3
2 3 3 3 5 8 8 8
\end{lstlisting}
\nextsample
\begin{lstlisting}
3 8
\end{lstlisting}
\end{makesample}
