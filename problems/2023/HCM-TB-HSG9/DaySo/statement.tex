\makeproblem{DaySo}{Dãy số (5đ)}{1.0s}{256 MB}

Cho dãy số nguyên $a_1, a_2, \dots, a_n$. Số $a_p\ (1 \le p \le n)$ được gọi là một số trung bình cộng trong dãy nếu tồn tại 3 chỉ số $i, j, k\ (1 \le i, j, k \le n)$ đôi một khác nhau, sao cho $a_p=(a_i+a_j+a_k) \div 3$.

\medskip
\textbf{Yêu cầu:} Cho $n$ và dãy số $a_1, a_2, \dots, a_n$. Hãy tìm số lượng các số trung bình cộng trong dãy.

\makesection{Input}
\begin{itemize}
    \item Dòng đầu chứa số nguyên dương $n\ (3 \le n \le 1000)$.
    \item Dòng thứ hai chứa $n$ số nguyên $a_i\ (a_i \le 10^{8})$
\end{itemize}

\makesection{Output}
\begin{itemize}
    \item In ra một số nguyên duy nhất là Số lượng các số trung bình cộng trong dãy.
\end{itemize}

\makesection{Ví dụ}

\subsubsection*{Ví dụ 1}
\begin{makesample}
\begin{lstlisting}
5
4 3 6 3 5
\end{lstlisting}
\nextsample
\begin{lstlisting}
2
\end{lstlisting}
\end{makesample}
